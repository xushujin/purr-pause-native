cmake_minimum_required(VERSION 3.22)
project(PurrPause VERSION 1.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  if(NOT CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET 12.0 CACHE STRING "Minimum macOS version" FORCE)
  endif()
endif()
include(GNUInstallDirs)
include(CTest)
option(PURR_DESKTOP_TESTS "Run Linux X11, window-manager, D-Bus and login-entry integration tests" OFF)
set(PURR_CODEC_PREFIX "" CACHE PATH "Self-contained Linux codec prefix built by scripts/build-linux-codecs.sh")
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Widgets Network Svg)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale)

add_library(purr_core STATIC src/core.cpp src/core.h src/messages.cpp src/messages.h)
target_include_directories(purr_core PUBLIC src)
target_link_libraries(purr_core PUBLIC Qt6::Core Qt6::Network OpenSSL::Crypto)
target_compile_definitions(purr_core PUBLIC PURR_VERSION="${PROJECT_VERSION}")

add_library(purr_ui STATIC src/media.cpp src/media.h src/overlays.cpp src/overlays.h src/dialogs.cpp src/dialogs.h)
target_link_libraries(purr_ui PUBLIC purr_core Qt6::Widgets Qt6::Svg PkgConfig::FFMPEG)
if(APPLE)
  target_sources(purr_ui PRIVATE src/platform_mac.mm)
  target_link_libraries(purr_ui PUBLIC "-framework AppKit" "-framework IOKit" "-framework CoreGraphics" "-framework CoreFoundation" "-framework CoreServices" "-framework ServiceManagement")
else()
  find_package(Qt6 REQUIRED COMPONENTS DBus)
  find_package(X11 REQUIRED)
  target_sources(purr_ui PRIVATE src/platform_linux.cpp)
  target_link_libraries(purr_ui PUBLIC Qt6::DBus X11::X11 X11::Xss)
endif()

add_executable(purr-pause MACOSX_BUNDLE src/main.cpp src/application.cpp src/application.h src/installation.cpp src/installation.h)
target_link_libraries(purr-pause PRIVATE purr_ui)
target_compile_definitions(purr-pause PRIVATE PURR_SOURCE_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/assets" PURR_SOURCE_DOCS="${CMAKE_CURRENT_SOURCE_DIR}/docs")
if(APPLE)
  set_target_properties(purr-pause PROPERTIES
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/Info.plist.in"
    MACOSX_BUNDLE_BUNDLE_NAME "PurrPause"
    MACOSX_BUNDLE_GUI_IDENTIFIER "com.purr-pause.app"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}")
  add_custom_command(TARGET purr-pause POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_BUNDLE_DIR:purr-pause>/Contents/Resources/assets"
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:purr-pause>/Contents/Resources/docs"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/docs/message-api-spec.md" "$<TARGET_BUNDLE_DIR:purr-pause>/Contents/Resources/docs/message-api-spec.md"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/assets/icon.icns" "$<TARGET_BUNDLE_DIR:purr-pause>/Contents/Resources/icon.icns")
  install(TARGETS purr-pause BUNDLE DESTINATION .)
else()
  if(PURR_CODEC_PREFIX)
    set_target_properties(purr-pause PROPERTIES INSTALL_RPATH "$ORIGIN/../lib/purr-pause")
    file(GLOB codec_libraries "${PURR_CODEC_PREFIX}/lib/libavcodec.so*" "${PURR_CODEC_PREFIX}/lib/libavformat.so*"
      "${PURR_CODEC_PREFIX}/lib/libavutil.so*" "${PURR_CODEC_PREFIX}/lib/libswscale.so*" "${PURR_CODEC_PREFIX}/lib/libvpx.so*")
    if(NOT codec_libraries)
      message(FATAL_ERROR "PURR_CODEC_PREFIX contains no codec libraries; run scripts/build-linux-codecs.sh first")
    endif()
    install(FILES ${codec_libraries} DESTINATION lib/purr-pause)
    install(DIRECTORY "${PURR_CODEC_PREFIX}/licenses/" DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/purr-pause/third-party)
  endif()
  install(TARGETS purr-pause RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(DIRECTORY assets DESTINATION ${CMAKE_INSTALL_DATADIR}/purr-pause)
  install(FILES docs/message-api-spec.md DESTINATION ${CMAKE_INSTALL_DATADIR}/purr-pause/docs)
  install(FILES packaging/purr-pause.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES assets/images/logo.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/512x512/apps RENAME purr-pause.png)
  install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/purr-pause)
endif()

if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  add_test(NAME macos_packaging COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_macos_packaging.py -v)
  set_tests_properties(macos_packaging PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 30)
  foreach(test_name core messages media ui instance dialogs application installation)
    add_executable(test_${test_name} tests/test_${test_name}.cpp)
    target_link_libraries(test_${test_name} PRIVATE purr_ui Qt6::Test)
    target_compile_definitions(test_${test_name} PRIVATE PURR_TEST_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/assets")
    target_compile_definitions(test_${test_name} PRIVATE PURR_TEST_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME ${test_name} COMMAND test_${test_name})
    set_tests_properties(${test_name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
  endforeach()
  target_sources(test_application PRIVATE src/application.cpp src/application.h)
  add_executable(test_visual tests/visual/capture_native.cpp)
  target_link_libraries(test_visual PRIVATE purr_ui)
  target_compile_definitions(test_visual PRIVATE PURR_TEST_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_definitions(test_instance PRIVATE PURR_TEST_EXECUTABLE="$<TARGET_FILE:purr-pause>")
  target_compile_definitions(test_installation PRIVATE PURR_TEST_EXECUTABLE="$<TARGET_FILE:purr-pause>")
  add_dependencies(test_instance purr-pause)
  add_dependencies(test_installation purr-pause)
  add_test(NAME app_smoke COMMAND purr-pause --smoke-test --data-dir "${CMAKE_CURRENT_BINARY_DIR}/smoke-data")
  set_tests_properties(app_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;XDG_DATA_HOME=${CMAKE_CURRENT_BINARY_DIR}/smoke-home" TIMEOUT 15)
  if(APPLE)
    add_executable(test_macos_platform tests/test_macos_platform.mm)
    set_target_properties(test_macos_platform PROPERTIES MACOSX_BUNDLE FALSE)
    target_link_libraries(test_macos_platform PRIVATE purr_ui Qt6::Test)
    target_compile_definitions(test_macos_platform PRIVATE PURR_TEST_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/assets")
    add_test(NAME macos_platform COMMAND test_macos_platform)
    set_tests_properties(macos_platform PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=cocoa" TIMEOUT 45)
    add_test(NAME cocoa_ui COMMAND test_ui)
    set_tests_properties(cocoa_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=cocoa" TIMEOUT 90)
  endif()
  if(PURR_DESKTOP_TESTS AND NOT APPLE)
    find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
    find_program(XVFB_RUN xvfb-run REQUIRED)
    find_program(OPENBOX openbox REQUIRED)
    find_program(GIO gio REQUIRED)
    if(NOT TARGET X11::Xtst)
      message(FATAL_ERROR "Desktop tests require libxtst-dev")
    endif()
    add_executable(test_platform tests/test_platform.cpp src/platform_linux.cpp)
    target_include_directories(test_platform PRIVATE src)
    target_link_libraries(test_platform PRIVATE Qt6::Widgets Qt6::DBus Qt6::Test X11::X11 X11::Xss X11::Xtst)
    add_test(NAME platform COMMAND "${DBUS_RUN_SESSION}" -- "${XVFB_RUN}" -a -s "-screen 0 1280x900x24" bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/run-desktop.sh" "$<TARGET_FILE:test_platform>")
    set_tests_properties(platform PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;XDG_SESSION_TYPE=x11" TIMEOUT 60)
    add_executable(test_screens tests/test_screens.cpp)
    target_link_libraries(test_screens PRIVATE purr_ui Qt6::Test)
    target_compile_definitions(test_screens PRIVATE PURR_TEST_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/assets")
    add_test(NAME screens COMMAND "${DBUS_RUN_SESSION}" -- "${XVFB_RUN}" -a -s "-screen 0 1280x900x24 -screen 1 1024x768x24" "$<TARGET_FILE:test_screens>")
    set_tests_properties(screens PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;XDG_SESSION_TYPE=x11" TIMEOUT 30)
  endif()
endif()
